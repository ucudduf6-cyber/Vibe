import {createClient} from 'https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2/+esm';
const $=id=>document.getElementById(id),cfg=window.WORKER_CONFIG||{},slotsKey='worker_accounts_v1';
const slots=()=>{try{return JSON.parse(localStorage.getItem(slotsKey)||'[]')}catch{return []}};
const setSlots=x=>localStorage.setItem(slotsKey,JSON.stringify(x));
const clients=new Map();let client=null,session=null,peer=null,poll=null,subscription=null;
function show(id){['setup','auth','app'].forEach(x=>$(x).classList.toggle('hide',x!==id))}
function fail(message,target='error'){$(target).textContent=message||''}
function currentSlot(){return localStorage.getItem('worker_current_slot')||'0'}
function makeClient(slot){if(!clients.has(slot))clients.set(slot,createClient(cfg.supabaseUrl,cfg.supabaseAnonKey,{auth:{storageKey:'worker_auth_'+slot,persistSession:true,autoRefreshToken:true,detectSessionInUrl:slot===currentSlot()}}));return clients.get(slot)}
function setCurrent(slot){localStorage.setItem('worker_current_slot',slot);client=makeClient(slot)}
async function start(){if(!/^https:\/\//.test(cfg.supabaseUrl||'')||!cfg.supabaseAnonKey){show('setup');return}let ss=slots();if(!ss.length){ss=['0'];setSlots(ss)}setCurrent(ss.includes(currentSlot())?currentSlot():ss[0]);await refreshAuth()}
async function refreshAuth(){const {data:{session:s}}=await client.auth.getSession();session=s;if(!session){show('auth');return}show('app');peer=null;$('messages').replaceChildren();$('peerName').textContent='Выбери собеседника';await loadMe();renderSlots();if(poll)clearInterval(poll);poll=setInterval(()=>{if(peer)loadMessages()},4000)}
async function loadMe(){const {data}=await client.from('profiles').select('username,display_name').eq('id',session.user.id).maybeSingle();$('me').textContent=data?`${data.display_name} (@${data.username})`:session.user.email;$('accountCount').textContent=`Аккаунтов на устройстве: ${slots().length}/3`}
function renderSlots(){$('switches').replaceChildren();slots().forEach(slot=>{let b=document.createElement('button');b.className='secondary';b.textContent=slot===currentSlot()?'● Аккаунт '+(Number(slot)+1):'Аккаунт '+(Number(slot)+1);b.onclick=async()=>{if(slot===currentSlot())return;await switchSlot(slot)};$('switches').append(b)})}
async function switchSlot(slot){if(subscription){await client.removeChannel(subscription);subscription=null}setCurrent(slot);await refreshAuth()}
$('signIn').onclick=async()=>{fail('','authError');const {error}=await client.auth.signInWithPassword({email:$('email').value.trim(),password:$('password').value});if(error)return fail(error.message,'authError');await refreshAuth()};
$('signUp').onclick=async()=>{fail('','authError');const email=$('email').value.trim(),password=$('password').value,username=$('username').value.trim().toLowerCase(),display_name=$('displayName').value.trim()||username;if(!/^[a-z0-9_]{5,32}$/.test(username))return fail('Юзернейм: 5–32 латинских символа, цифры и _.','authError');if(password.length<8)return fail('Пароль должен содержать минимум 8 символов.','authError');const {data,error}=await client.auth.signUp({email,password});if(error)return fail(error.message,'authError');if(!data.session)return fail('Проверь почту и подтверди email, затем войди.','authError');const {error:e}=await client.from('profiles').insert({id:data.user.id,username,display_name});if(e)return fail('Аккаунт создан, но юзернейм не сохранён: '+e.message,'authError');await refreshAuth()};
$('addAccount').onclick=async()=>{let ss=slots();if(ss.length>=3)return fail('Максимум 3 аккаунта. Сначала выйди из одного.');let next=['0','1','2'].find(x=>!ss.includes(x));ss.push(next);setSlots(ss);await switchSlot(next)};
$('logout').onclick=async()=>{if(subscription){await client.removeChannel(subscription);subscription=null}if(poll){clearInterval(poll);poll=null}await client.auth.signOut();let ss=slots().filter(x=>x!==currentSlot());if(!ss.length)ss=['0'];setSlots(ss);await switchSlot(ss[0])};
$('find').onclick=async()=>{fail('');let u=$('search').value.trim().toLowerCase().replace(/^@/,'');if(!u)return;const {data,error}=await client.from('profiles').select('id,username,display_name').eq('username',u).maybeSingle();if(error)return fail(error.message);$('people').replaceChildren();if(!data)return fail('Пользователь не найден');if(data.id===session.user.id)return fail('Это твой аккаунт');const b=document.createElement('button');b.className='secondary';b.textContent=`${data.display_name} (@${data.username})`;b.onclick=()=>openPeer(data);$('people').append(b)};
async function openPeer(data){peer=data;$('peerName').textContent=`${data.display_name} (@${data.username})`;$('block').classList.remove('hide');await loadMessages();if(subscription){await client.removeChannel(subscription);subscription=null}subscription=client.channel('dm-'+session.user.id+'-'+peer.id).on('postgres_changes',{event:'INSERT',schema:'public',table:'direct_messages'},payload=>{if(payload.new.sender_id===peer?.id||payload.new.recipient_id===peer?.id)loadMessages()}).subscribe();await updateBlock()}
async function loadMessages(){if(!peer||!session)return;const p=peer.id,my=session.user.id;const {data,error}=await client.from('direct_messages').select('id,sender_id,recipient_id,body,created_at').or(`and(sender_id.eq.${my},recipient_id.eq.${p}),and(sender_id.eq.${p},recipient_id.eq.${my})`).order('created_at',{ascending:false}).limit(100);if(error)return fail(error.message);const box=$('messages');box.replaceChildren();(data||[]).reverse().forEach(m=>{let d=document.createElement('div');d.className='bubble'+(m.sender_id===my?' mine':'');d.textContent=m.body;box.append(d)});box.scrollTop=box.scrollHeight}
$('compose').onsubmit=async e=>{e.preventDefault();if(!peer)return fail('Сначала выбери собеседника');let body=$('message').value.trim();if(!body)return;const {error}=await client.from('direct_messages').insert({sender_id:session.user.id,recipient_id:peer.id,body});if(error)return fail('Не отправлено: '+error.message);$('message').value='';fail('');await loadMessages()};
async function updateBlock(){if(!peer)return;const {data}=await client.from('blocks').select('blocked_id').eq('blocker_id',session.user.id).eq('blocked_id',peer.id).maybeSingle();$('block').textContent=data?'Разблокировать':'Заблокировать'}
$('block').onclick=async()=>{if(!peer)return;const {data}=await client.from('blocks').select('blocked_id').eq('blocker_id',session.user.id).eq('blocked_id',peer.id).maybeSingle();const q=client.from('blocks');const {error}=data?await q.delete().eq('blocker_id',session.user.id).eq('blocked_id',peer.id):await q.insert({blocker_id:session.user.id,blocked_id:peer.id});if(error)return fail(error.message);await updateBlock()};
start().catch(e=>{show('auth');fail(e.message,'authError')});
